# Release: maximum optimization, no debug data, stripped for distribution.
# Use for final binaries, benchmarks against baselines, and published artifacts.
[profile.release]
opt-level = 3
debug = false               # no debug info in production binaries
strip = "symbols"           # remove symbol table for smaller binary
lto = "fat"                 # full cross-crate LTO (slowest compile, best runtime)
codegen-units = 1           # single codegen unit = maximum LLVM optimization
overflow-checks = true      # keep arithmetic behavior aligned with dev/test lanes
panic = "unwind"            # required: quantum_core/src/gpu/peps.rs:204 uses catch_unwind

# Gate: fast-compile release for governance gate binaries (TOML parsing + SHA-256).
# Why not full release: fat LTO + 1 codegen-unit takes 2-9 minutes for binaries
# that spend 90% of runtime in TOML deserialization. Thin LTO + 6 codegen-units
# compiles in ~15s and runs within 10% of fat-LTO speed for I/O-bound workloads.
[profile.release-gate]
inherits = "release"
lto = "thin"                # 10x faster link vs fat LTO; <5% runtime difference for TOML parsing
codegen-units = 6           # parallel codegen; matches physical core count
debug = "line-tables-only"  # perf/flamegraph profiling without full DWARF
strip = "none"              # keep symbols for perf report
overflow-checks = true

# Parity: bit-exact numerical match with release for validation.
# Uses LLVM (codegen-backend = "llvm" overrides dev's cranelift).
# inherits = "dev" propagates cranelift; must explicitly override here.
# Use: cargo build --profile=parity -p <crate>
# Use: cargo test --profile=parity -p <crate>
[profile.parity]
