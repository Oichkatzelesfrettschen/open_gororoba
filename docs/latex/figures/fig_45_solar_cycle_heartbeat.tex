%% Fig 45: Solar Cycle CD Heartbeat -- 23-year ACE MAG at 1 AU
%% Dual y-axis: CD median (left, log) + sunspot proxy (right, linear)
%% Shows 18.9x solar cycle amplitude with phase markers for SC 23/24/25
%%
%% Data: ACE MAG 16-sec via HAPI, 45.4M records, 32D 4-channel embedding

%%
%% Golden ratio: width/height = 1.618
\begin{tikzpicture}
\begin{axis}[
  name=cdaxis,
  width=\textwidth,
  height=0.618\textwidth,
