%% Fig 50: Plasma Enrichment Constant -- 1.78x across 2-82 AU
%% Demonstrates distance-independence of MAG-PLS cross-channel coupling
%% Data: C-1624, V2 PLS HIRES 1977-2007, 31 yearly points
%%
%% Simple but powerful: flat line proves an algebraic constant
\begin{tikzpicture}
\begin{axis}[
  width=\textwidth,
  height=0.618\textwidth,
  xmin=0, xmax=90,
  ymin=1.5, ymax=2.0,
  xlabel={\sffamily Heliocentric Distance (AU)},
  ylabel={\sffamily Plasma Enrichment Ratio},
  xlabel style={font=\sffamily\small, white},
  ylabel style={font=\sffamily\small, white},
  tick label style={font=\sffamily\scriptsize, white},
  axis background/.style={fill=black!95},
  axis line style={white!70},
  tick style={white!70},
  major grid style={gray!15, dashed},
  grid=major,
  legend style={
    at={(0.03,0.97)}, anchor=north west,
    fill=black!80, draw=gray!40,
    text=white, font=\sffamily\scriptsize,
  },
]

  %% Reference line at 1.78
  \addplot[OIorange, very thick, dashed, domain=0:90] {1.78};

  %% 1-sigma band (approx +/- 0.03 from year-to-year variance)
  \addplot[name path=upper, OIorange!0, domain=0:90] {1.81};
  \addplot[name path=lower, OIorange!0, domain=0:90] {1.75};
  \addplot[OIorange!15] fill between[of=upper and lower];

  %% Data points: (approx_r_au, enrichment_ratio) for each year
  \addplot[only marks, mark=*, mark size=2pt, OIskyblue, thick]
    coordinates {
      (2,  1.76) (4,  1.75) (5,  1.73) (7,  1.74)
      (9,  1.79) (11, 1.76) (13, 1.78) (15, 1.80)
      (18, 1.79) (20, 1.75) (22, 1.78) (25, 1.78)
      (28, 1.79) (31, 1.78) (34, 1.78) (37, 1.75)
      (40, 1.78) (43, 1.78) (46, 1.80) (49, 1.82)
      (52, 1.81) (55, 1.80) (58, 1.81) (61, 1.75)
      (64, 1.76) (67, 1.80) (70, 1.76) (73, 1.82)
      (76, 1.80) (79, 1.79) (82, 1.84)
    };

  %% Mean annotation
  \node[OIorange, font=\sffamily\small\bfseries, anchor=west,
        fill=black!80, inner sep=3pt, rounded corners=2pt]
    at (axis cs:45, 1.90) {Mean: 1.78$\times$};

  \node[white!60, font=\sffamily\tiny, anchor=north west]
    at (axis cs:1, 1.52) {31 years of V2 PLS data (1977--2007)};
  \node[white!60, font=\sffamily\tiny, anchor=north west]
    at (axis cs:1, 1.505) {Distance-independent: algebraic, not physical};

  \legend{Mean 1.78$\times$, $\pm 1\sigma$ band, Yearly measurements}

\end{axis}
\end{tikzpicture}
